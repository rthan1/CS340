# --- While loop with nested if ---
integer i = 1
integer sum = 0

# Sum of 1..5, printing the running total (1, 3, 6, 10, 15)
while (i <= 5):
    sum = sum + i
    if (i % 2 == 0):
        # Even i: add 0 (no effect, just to exercise nested if)
        sum = sum + 0
    else:
        # Odd i: also no-op, just another branch
        sum = sum + 0
    i = i + 1

print(sum)


# --- If / elif / else chain (reuses sum) ---
integer grade = 0

if (sum >= 20):
    grade = 1
elif (sum >= 15):
    grade = 2
elif (sum >= 10):
    grade = 3
else:
    grade = 4

print(grade)

# --- Functions with control flow and nested blocks ---

# Absolute value using if/else
func abs_val(x):
    if (x < 0):
        return 0 - x
    else:
        return x


# Function that uses while + nested if/elif/else and prints from inside
func classify_and_sum(limit):
    integer i2 = 0
    integer total = 0

    while (i2 <= limit):
        if (i2 == 0):
            # print 0 for zero
            print(0)
        elif (i2 % 2 == 0):
            # even > 0 -> print 2
            print(2)
        else:
            # odd -> print 1
            print(1)

        total = total + i2
        i2 = i2 + 1

    return total


# Function that uses another function and nested conditionals
func nested_funcs(a1, b1):
    integer av1 = 0
    integer av2 = 0

    av1 = abs_val(a1)
    av2 = abs_val(b1)

    if (av1 > av2):
        print(100)
    elif (av1 == av2):
        print(200)
    else:
        print(300)

    return av1 + av2

integer main_result = 0

# classify_and_sum(3) should:
